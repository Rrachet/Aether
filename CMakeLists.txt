cmake_minimum_required(VERSION 3.20)
project(aether LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AETHER_BUILD_TESTS "Build tests" ON)
option(AETHER_BUILD_GGUF_TESTS "Build the GGUF header tests (fail until docs/GGUF_MILESTONE.md is done)" OFF)
add_library(aether STATIC src/tensor.cpp src/runtime.cpp src/gguf.cpp)
target_include_directories(aether PUBLIC include)
target_compile_options(aether PRIVATE -Wall -Wextra -Wpedantic)
add_executable(aether-run src/main.cpp)
target_link_libraries(aether-run PRIVATE aether)
if(AETHER_BUILD_TESTS)
 enable_testing()
 add_executable(aether-test tests/test_tensor.cpp)
 target_link_libraries(aether-test PRIVATE aether)
 add_test(NAME aether-test COMMAND aether-test)
 if(AETHER_BUILD_GGUF_TESTS)
  add_executable(aether-gguf-test tests/test_gguf.cpp)
  target_link_libraries(aether-gguf-test PRIVATE aether)
  add_test(NAME aether-gguf-test COMMAND aether-gguf-test)
 endif()
endif()
